// Submit Cargo Search on airsellcargo.com async function searchCargoRates() { const searchParams = { origin: document.getElementById('originAirport').value, // e.g. "HGA" (Hargeisa) destination: document.getElementById('destAirport').value, // e.g. "DXB" (Dubai) weight: document.getElementById('cargoWeight').value, // e.g. 250 (kg) volume: document.getElementById('cargoVolume').value // e.g. 1.5 (m³) }; try { const response = await fetch('https://api.airsellcargo.com/api/cargo/search', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(searchParams) }); const result = await response.json(); if (result.success) { console.log('Available Rates:', result.data); // Function to render rates on your web page displayRatesOnUI(result.data); } else { alert('Error fetching cargo rates: ' + result.error); } } catch (err) { console.error('Connection error:', err); } }
